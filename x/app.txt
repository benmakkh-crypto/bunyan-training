function renderQ(q){
  const n = 'n'+q.id.replace(/\W/g,'_');
  let body='';
  if(q.type==='نص') body = `<textarea data-q="${esc(q.id)}" rows="3"></textarea>`;
  else if(q.type==='مقياس') body = `<div class="scale">${q.options.map(o=>`<label><input type="radio" name="${n}" data-q="${esc(q.id)}" value="${esc(o)}">${esc(o)}</label>`).join('')}</div>`;
  else if(q.type==='متعدد') body = q.options.map(o=>`<label class="opt"><input type="checkbox" data-q="${esc(q.id)}" data-multi="1" value="${esc(o)}">${esc(o)}</label>`).join('');
  else body = q.options.map(o=>`<label class="opt"><input type="radio" name="${n}" data-q="${esc(q.id)}" value="${esc(o)}">${esc(o)}</label>`).join('');
  return `<div class="q"><p>${esc(q.text)}${q.type==='نص'?` <span class="mu">${t('optional','(اختياري)')}</span>`:''}</p>${body}</div>`;
}
function collect(){
  const a={};
  document.querySelectorAll('[data-q]').forEach(el=>{
    const id=el.dataset.q;
    if(el.tagName==='TEXTAREA'){ if(el.value.trim()) a[id]=el.value.trim(); }
    else if(el.dataset.multi){ a[id]=a[id]||[]; if(el.checked) a[id].push(el.value); }
    else if(el.checked) a[id]=el.value;
  });
  return a;
}

function capDraw(id){
  const code = String(Math.floor(1000 + Math.random()*9000)); CAP[id] = code;
  const c = document.getElementById('cap_'+id); if(!c) return; const x = c.getContext('2d');
  const g = x.createLinearGradient(0,0,120,44); g.addColorStop(0,'#26407F'); g.addColorStop(1,'#3D5BA9'); x.fillStyle = g; x.fillRect(0,0,120,44);
  for(let i=0;i<5;i++){ x.strokeStyle='rgba(255,255,255,.25)'; x.beginPath(); x.moveTo(Math.random()*120, Math.random()*44); x.bezierCurveTo(Math.random()*120,Math.random()*44,Math.random()*120,Math.random()*44,Math.random()*120,Math.random()*44); x.stroke(); }

  code.split('').forEach((d,i)=>{ x.save(); x.translate(12+i*25, 23+(Math.random()*8-4)); x.rotate((Math.random()-.5)*.6); x.fillText(d,0,0); x.restore(); });
}

function cvText(x){
  const fit = (s, size, font, w, max) => { let z=size; x.font=`${w} ${z}px ${font}`; while(max && x.measureText(s).width>max && z>8){ z-=.5; x.font=`${w} ${z}px ${font}`; } return z; };

}

function code128(txt){
  const v = String(txt).split('').map(ch => Math.max(0, Math.min(94, ch.charCodeAt(0) - 32)));
  let sum = 104; v.forEach((x,i) => sum += x * (i+1));

}

const ST_COLORS = {'تم التنفيذ':'#1F8A5B','جاري التنفيذ':'#B87410','لم يبدأ تنفيذه':'#3D5BA9','متعثر':'#C0392B','مؤجل':'#7A3FB0','ملغى':'#A0522D'};
const fmtN = n => Number(n||0).toLocaleString('en');
function niceMax(v){ const p=Math.pow(10,Math.floor(Math.log10(v))); const m=v/p; return (m<=1?1:m<=2?2:m<=2.5?2.5:m<=5?5:10)*p; }
function colChart(data, o){
  // data: [{label, values:[...], tip}] · o.series: [{name,color}]
  const ns=o.series.length, bwMax=ns>1?22:34, gap=ns>1?2:0;
  const H=o.h||290, pt=26, pb=78, pr=50, pl=8;
  const W=Math.max(o.minW||560, data.length*(ns*(bwMax+gap)+22)+pr+pl);
  const ih=H-pt-pb, iw=W-pr-pl, gw=iw/Math.max(data.length,1);
  const max=niceMax(Math.max(1,...data.flatMap(d=>d.values)));
  const bw=Math.min(bwMax,(gw-14)/ns-gap), y=v=>pt+ih-(v/max)*ih;
  let g='';
  for(let i=0;i<=4;i++){ const v=max*i/4, yy=y(v); g+=`<line x1="${pl}" x2="${W-pr}" y1="${yy}" y2="${yy}" stroke="#E3E8F1" stroke-width="1"/><text x="${W-pr+8}" y="${yy+4}" font-size="11" fill="#5F6676" text-anchor="start">${fmtN(v)}</text>`; }
  data.forEach((d,i)=>{
    const cx = W-pr-(i+0.5)*gw, x0 = cx - (ns*bw+(ns-1)*gap)/2;
    d.values.forEach((v,j)=>{ const x=x0+(ns-1-j)*(bw+gap), yy=y(v), h=Math.max(0,pt+ih-yy), r=Math.min(4,h);
      g+=`<path d="M${x},${pt+ih} V${yy+r} Q${x},${yy} ${x+r},${yy} H${x+bw-r} Q${x+bw},${yy} ${x+bw},${yy+r} V${pt+ih} Z" fill="${o.series[j].color}"/>`;
      if(ns===1 && v>0) g+=`<text x="${x+bw/2}" y="${yy-6}" font-size="12" font-weight="700" fill="#1E2433" text-anchor="middle">${fmtN(v)}</text>`; });
    g+=`<rect x="${cx-gw/2}" y="${pt}" width="${gw}" height="${ih}" fill="transparent" data-tip="${esc(d.tip||'')}"/>`;
    const lbl = String(d.label).length>14 ? String(d.label).slice(0,13)+'…' : d.label;
    g+=`<text x="${cx}" y="${pt+ih+16}" font-size="12.5" fill="#1E2433" text-anchor="end" transform="rotate(-35 ${cx} ${pt+ih+16})">${esc(lbl)}</text>`;
  });
  g+=`<line x1="${pl}" x2="${W-pr}" y1="${pt+ih}" y2="${pt+ih}" stroke="#9AA3B5"/>`;
  const legend = ns>1 ? `<div class="legend row2">${o.series.map(sr=>`<div><i style="background:${sr.color}"></i>${esc(sr.name)}</div>`).join('')}</div>` : '';
  return legend + `<div class="cscroll"><svg width="100%" viewBox="0 0 ${W} ${H}" style="min-width:${Math.min(W,520)}px;height:auto;direction:ltr;font-family:'IBM Plex Sans Arabic',sans-serif" role="img" aria-label="${esc(o.title||'')}">${g}</svg></div>`;
}
function donut(rows, centerLabel){
  // rows: [[label, value, color]]
  const tot = rows.reduce((a,r)=>a+r[1],0)||1, R=74, C=2*Math.PI*R, S=200;
  let off=0, arcs='';
  rows.forEach(([l,v,c])=>{ const len=v/tot*C, d=Math.max(0,len-2);
    arcs+=`<circle cx="${S/2}" cy="${S/2}" r="${R}" fill="none" stroke="${c}" stroke-width="30" stroke-dasharray="${d} ${C-d}" stroke-dashoffset="${-off}" transform="rotate(-90 ${S/2} ${S/2})" data-tip="${esc(l)}: ${fmtN(v)} (${Math.round(v/tot*100)}%)"/>`; off+=len; });
  return `<div class="donut-wrap"><svg width="${S}" height="${S}" viewBox="0 0 ${S} ${S}" role="img" aria-label="${esc(centerLabel)}">${arcs}
    <text x="${S/2}" y="${S/2+4}" text-anchor="middle" font-size="34" font-weight="700" fill="#1E2433" font-family="IBM Plex Sans Arabic,sans-serif">${fmtN(tot)}</text>
    <text x="${S/2}" y="${S/2+28}" text-anchor="middle" font-size="13" fill="#5F6676" font-family="IBM Plex Sans Arabic,sans-serif">${esc(centerLabel)}</text></svg>
    <div class="legend" style="flex:1;min-width:180px">${rows.map(([l,v,c])=>`<div><i style="background:${c}"></i>${esc(l)}<b>${fmtN(v)} · ${Math.round(v/tot*100)}%</b></div>`).join('')}</div></div>`;
}

async function uploadVideo(file, onPct){
  const base = {cid:EV.cid, date:EV.date||today()};
  try{
    const s = await run('evVideoSession', TOK, Object.assign({name:file.name, mime:file.type||'video/mp4', size:file.size, origin:location.origin}, base));
    const id = await new Promise((res,rej)=>{ const x=new XMLHttpRequest(); x.open('PUT', s.uploadUrl);
      x.upload.onprogress = e => { if(e.lengthComputable) onPct(Math.round(e.loaded/e.total*100)); };
      x.onload = () => { if(x.status>=200&&x.status<300){ try{ res(JSON.parse(x.responseText).id); }catch(er){ rej(er); } } else rej(new Error('status '+x.status)); };
      x.onerror = () => rej(new Error('network')); x.send(file.slice(0,file.size,'')); });
    return await run('evUploadDone', TOK, {id, key:file.name+'|'+file.size+'|'+file.lastModified});
  }catch(e){
    if(String(e).includes('مرفوع سابقاً')) throw e;
    if(file.size <= 28*1024*1024){ onPct(50); const data = await toB64(file); return await run('evUpload', TOK, Object.assign({kind:'video', name:file.name, mime:file.type||'video/mp4', data}, base)); }
    throw new Error('تعذّر رفع الفيديو (الحجم كبير أو الاتصال ضعيف) — جرّب مقطعاً أقصر');
  }
}
async function onSlotFile(input){
  const k=input.dataset.k, i=+input.dataset.i, file=input.files[0]; if(!file) return; input.value='';
  if(!EV.cid){ toast('اختر البرنامج أولاً'); return; }
  if(k==='photos' && !file.type.startsWith('image/')){ EV[k][i]={state:'err',msg:'الملف ليس صورة'}; return drawSlots(); }
  if(k==='videos' && !file.type.startsWith('video/')){ EV[k][i]={state:'err',msg:'الملف ليس فيديو'}; return drawSlots(); }
  const key = k==='photos' ? await sha256(file) : file.name+'|'+file.size+'|'+file.lastModified;
  if(EV[k].some((s,j)=>j!==i && s && s.key===key)){ EV[k][i]={state:'err',msg:'هذا الملف مكرر في الشاهد نفسه'}; return drawSlots(); }
  const slot = {state:'up', pct:0, key, name:file.name}; EV[k][i]=slot; EV.busy++; drawSlots();
  try{
    if(k==='photos'){
      slot.pct=30; drawSlots(); let f=file;
      if(f.size>1.2e6 && !/heic|heif/.test(f.type)){ const img=await createImageBitmap(f).catch(()=>null);
        if(img){ const sc=Math.min(1,1800/Math.max(img.width,img.height)), c=document.createElement('canvas'); c.width=img.width*sc; c.height=img.height*sc; c.getContext('2d').drawImage(img,0,0,c.width,c.height);
          const b=await new Promise(r=>c.toBlob(r,'image/jpeg',.82)); if(b) f=new File([b], file.name.replace(/\.\w+$/,'')+'.jpg', {type:'image/jpeg'}); } }
      const data = await toB64(f); slot.pct=70; drawSlots();
      slot.file = await run('evUpload', TOK, {kind:'photo', cid:EV.cid, date:EV.date||today(), name:f.name, mime:f.type, data});
      slot.preview = URL.createObjectURL(f);
    } else slot.file = await uploadVideo(file, p=>{ slot.pct=p; drawSlots(); });
    slot.state='ok';
  }catch(e){ slot.state='err'; slot.msg=String(e); }
  finally{ EV.busy--; drawSlots(); }
}

// تقويم أيام التنفيذ: مكتمل / بلا شاهد / إجازة / قادم
function evCalendar(cid, list){
  const c = D.courses.find(x=>x['المعرف']===cid); if(!c) return '';
  const p = s => { const m=String(s||'').match(/(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})/); return m?new Date(+m[1],+m[2]-1,+m[3]):null; };
  let s = p(c['البدء']), e = p(c['الانتهاء']) || s;
  if(!s) return '<p class="mu">لا يوجد تاريخ بدء لهذا البرنامج لعرض التقويم.</p>';
  if(e < s) e = s; if((e-s)/864e5 > 120) e = new Date(s.getTime()+120*864e5);
  const done = new Set(list.map(x=>{ const d=p(x.date); return d?d.toDateString():''; }));
  const offDay = {'الجمعة':5,'السبت':6,'الأحد':0}[D.evSettings.off]; const now = new Date(); now.setHours(0,0,0,0);
  let h = '<div class="cal" style="margin-top:12px">' + ['أحد','إثنين','ثلاثاء','أربعاء','خميس','جمعة','سبت'].map(d=>`<div class="h">${d}</div>`).join('');
  for(let i=0;i<s.getDay();i++) h += '<div class="d-out"></div>';
  let okN=0, miss=0;
  for(let d=new Date(s); d<=e; d.setDate(d.getDate()+1)){
    const k = d.toDateString(), lbl = d.getDate()+'/'+(d.getMonth()+1);
    let cls = 'd-miss';
    if(done.has(k)){ cls='d-ok'; okN++; } else if(d.getDay()===offDay) cls='d-off'; else if(d>now) cls='d-fut'; else miss++;
    h += `<div class="${cls}">${lbl}</div>`;
  }
  return h + `</div><p class="mu">🟩 موثّق ${okN} · 🟥 بلا شاهد ${miss} · ⬜ إجازة/قادم</p>`;
}
